///////////////////////////////////////////////////////////////////////////////////////////////////////////

Listar eventos para o aluno/turma:

select *
from evento_escopo
where id_instituicao = :instituicao
  and (
    :turma is null
    or para_toda_escola
    or exists (
      select 1 from evento_turma et
      where et.id_evento = evento_escopo.id_evento
        and et.id_turma = :turma
    )
  );

///////////////////////////////////////////////////////////////////////////////////////////////////////////

Mostrar as turmas de cada evento:

select e.id_evento, e.nome,
       coalesce(
         string_agg(s.nome || ' ' || c.nome, ', ' order by s.nome),
         'Toda a escola'
       ) as destinatarios
from evento e
left join evento_turma et on et.id_evento = e.id_evento
left join turma t on t.id_turma = et.id_turma
left join serie s on s.id_serie = t.id_serie
left join curso c on c.id_curso = t.id_curso
where e.id_instituicao = :instituicao
group by e.id_evento, e.nome;
